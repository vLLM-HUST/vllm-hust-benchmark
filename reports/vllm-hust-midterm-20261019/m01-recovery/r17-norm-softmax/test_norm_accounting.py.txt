"""Independent scalar counts and vector RMS checks, not device BF16 checks."""

from collections import Counter
import math

import numpy as np
import pytest

from audit_norm_accounting import catalog, rms_operations


@pytest.mark.parametrize("width", [1, 2, 3, 127, 128, 256, 2048])
@pytest.mark.parametrize("rows", [0, 1, 3])
@pytest.mark.parametrize("residual", [False, True])
@pytest.mark.parametrize("gated", [False, True])
def test_scalar_math_and_counts(width, rows, residual, gated):
    rng = np.random.default_rng(27)
    x, r, z = [rng.normal(size=(rows, width)) for _ in range(3)]
    weight = rng.normal(size=width)
    counts = Counter()

    def op(name, fn, *args):
        counts[name] += 1
        return fn(*args)

    actual = []
    for row in range(rows):
        values = [op("residual_add", lambda a, b: a + b, a, b) if residual else a
                  for a, b in zip(x[row], r[row])]
        squares = [op("square", lambda a: a * a, a) for a in values]
        total = squares[0]
        for value in squares[1:]:
            total = op("reduction_add", lambda a, b: a + b, total, value)
        mean = op("mean_scale", lambda a: a / width, total)
        variance = op("epsilon_add", lambda a: a + 1e-6, mean)
        scale = op("inverse_sqrt_evaluations", lambda a: 1 / math.sqrt(a), variance)
        output = []
        for col, value in enumerate(values):
            value = op("normalize_multiply", lambda a: a * scale, value)
            value = op("weight_multiply", lambda a: a * weight[col], value)
            if gated:
                gate = op("silu_evaluations", lambda a: a / (1 + math.exp(-a)), z[row, col])
                value = op("output_gate_multiply", lambda a, b: a * b, value, gate)
            output.append(value)
        actual.append(output)
    expected_counts = rms_operations(rows, width, residual, gated)
    assert {key: counts[key] for key in expected_counts} == expected_counts
    if rows:
        values = x + r if residual else x
        expected = values / np.sqrt(np.mean(values ** 2, axis=-1, keepdims=True) + 1e-6) * weight
        if gated:
            expected *= z / (1 + np.exp(-z))
        np.testing.assert_allclose(actual, expected, rtol=1e-12, atol=1e-12)


@pytest.mark.parametrize("rows,width", [(-1, 128), (1, 0), (1.0, 128), (True, 128)])
def test_invalid_dimensions(rows, width):
    with pytest.raises(ValueError):
        rms_operations(rows, width)


def test_no_tp_duplicate_hidden_norm():
    entries = {e["name"]: e for e in catalog(1, 1)}
    assert entries["target.hidden_with_residual"]["rows"] == 80
    assert entries["target.hidden_no_residual"]["rows"] == 1
    assert entries["draft.embedding_hidden_and_first_norm"]["rows"] == 3
    assert entries["draft.post_attention_and_final_norm"]["rows"] == 2
    assert entries["target.full_attention_qk"]["rows"] == 180
    assert entries["draft.full_attention_qk"]["rows"] == 18
    assert entries["target.gdn_output"]["rows"] == 960
