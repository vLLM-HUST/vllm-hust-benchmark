   307		return e.scaleClient.Scales(scaledObject.Namespace).Get(ctx, scaledObject.Status.ScaleTargetGVKR.GroupResource(), scaledObject.Spec.ScaleTargetRef.Name, metav1.GetOptions{})
   308	}

   310	func (e *scaleExecutor) updateScaleOnScaleTarget(ctx context.Context, scaledObject *kedav1alpha1.ScaledObject, replicas int32) (int32, error) {
   311		pollingInterval, err := getPollingInterval(scaledObject)
   312		if err != nil {
   313			return -1, err
   314		}
   315		operationCtx, cancel := context.WithTimeout(ctx, pollingInterval+e.kubernetesAPITimeout)
   316		defer cancel()
   317		scale, err := e.getScaleTargetScale(operationCtx, scaledObject)
   318		if err != nil {
   319			return -1, err
   320		}
   321	
   322		// Update with requested replicas.
   323		currentReplicas := scale.Spec.Replicas
   324		scale.Spec.Replicas = replicas
   325	
   326		_, err = e.scaleClient.Scales(scaledObject.Namespace).Update(operationCtx, scaledObject.Status.ScaleTargetGVKR.GroupResource(), scale, metav1.UpdateOptions{})
   327		return currentReplicas, err
   328	}
